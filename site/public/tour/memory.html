Memory and allocators

Stack.raz
module Stack;
import Out, Mem, Heap, Arena;

type Node = record next: ^Node; val: INT; end;

(* a node holding v on top of top, from the allocator a;
   nil if a has no room *)
proc push(top: ^Node; v: INT; a: ^Mem.Allocator): ^Node;
var n: ^Node;
begin
    NEW(n, a);
    if n # nil then
        n^.next := top;
        n^.val := v;
    end;
    return n;
end;

proc main*;
var top, n: ^Node; i: INT; ar: Arena.Arena;
begin
    for i := 1 to 3 do top := push(top, i, Heap.heap); end;
    while top # nil do
        Out.int(top^.val); Out.char(' ');
        n := top;
        top := top^.next;
        Mem.free(Heap.heap, ^U8(n));
    end;
    Out.ln;
    if not Arena.init(ADR(ar), 100000) then return; end;
    for i := 1 to 1000 do top := push(top, i, ADR(ar.base)); end;
    Out.int(top^.val); Out.ln;
    Arena.done(ADR(ar));            (* every node at once *)
end;

end Stack.

raz has no garbage collector, and no allocator of its own: memory comes from an allocator, a Mem.Allocator, which the program chooses and passes.

NEW(n, a) sets n to a fresh, zeroed Node from a, or to nil if a has no room. Heap.heap is the general one, which Mem.free gives memory back to. An Arena hands out memory from one block and frees it all at once, as the last lines do.

Buf, List and Map take the allocator they use from init.

Save a program under its module’s name and run it with raz run Name.raz.
