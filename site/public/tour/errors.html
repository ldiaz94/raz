Errors

Errors.raz
module Errors;
import Out, Str;

const
    Empty*: ERROR = "nothing to read";
    TooBig*: ERROR = "not a number up to 1000";

(* the number s holds, up to 1000 *)
proc number(s: []U8): (INT, ERROR);
var n: I64; ok: BOOL;
begin
    if LEN(s) = 0 then return 0, Empty; end;
    n, ok := Str.toInt(s);
    if not ok or (n > 1000) then return 0, TooBig; end;
    return INT(n), nil;
end;

proc main*;
var words: [3][]U8; i, n: INT; err: ERROR;
begin
    words := ["42", "", "12345"];
    for i := 0 to 2 do
        n, err := number(words[i]);
        if err = nil then Out.int(n);
        elsif err = Empty then Out.str("empty");
        else Out.str(MSG(err));
        end;
        Out.ln;
    end;
end;

end Errors.
42
empty
not a number up to 1000

A procedure that can fail returns an ERROR beside its result, and nil when it didn't. Each ERROR constant is an error of its own, compared with = by which declaration made it, never by its message; MSG(err) is the message.

An exported one, as Empty is here, lets the callers tell it from the others. raz has no exceptions.

Save a program under its module’s name and run it with raz run Name.raz.
