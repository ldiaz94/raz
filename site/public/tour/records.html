Records and pointers

Boxes.raz
module Boxes;
import Fmt;

type
    Point = record x, y: INT; end;
    Box = record lo, hi: Point; end;

proc area(b: Box): INT;
begin
    return (b.hi.x - b.lo.x) * (b.hi.y - b.lo.y);
end;

proc grow(b: ^Box; n: INT);
begin
    DEC(b^.lo.x, n);
    DEC(b^.lo.y, n);
    INC(b^.hi.x, n);
    INC(b^.hi.y, n);
end;

proc main*;
var b: Box; p: ^Point;
begin
    b := {lo: {x: 1, y: 1}, hi: {x: 4, y: 3}};
    Fmt.print("{}\n", [Fmt.int(area(b))]);
    grow(ADR(b), 1);
    p := ADR(b.hi);
    p^.x := 10;
    Fmt.print("{} {}\n", [Fmt.int(area(b)), Fmt.int(b.hi.x)]);
end;

end Boxes.

A record is laid out as C lays out a struct. {lo: ..., hi: ...} builds one, naming its fields; the ones it leaves out are zero.

^Box is a pointer to a Box, ADR(b) the address of b, and b^ what a pointer points at. A pointer is never the record: it is b^.lo, never b.lo.
