Defer

Defers.raz
module Defers;
import Out, Sys;

proc step(n: INT);
begin
    Out.str("in "); Out.int(n); Out.ln;
    defer Out.str("out\n");
    if n = 2 then return; end;
    defer Out.str("done\n");
    Out.str("working\n");
end;

(* the lines of the file at path, or -1 if it can't be read *)
proc lines(path: []U8): INT;
var fd, n, i, count: INT; buf: [4096]U8;
begin
    fd := Sys.openRead(path);
    if fd < 0 then return -1; end;
    defer Sys.close(fd);
    count := 0;
    n := Sys.read(fd, buf);
    while n > 0 do
        for i := 0 to n - 1 do
            if buf[i] = '\n' then INC(count); end;
        end;
        n := Sys.read(fd, buf);
    end;
    return count;
end;

proc main*;
begin
    step(1);
    step(2);
    Out.int(lines("/dev/null")); Out.ln;
    Out.int(lines("/no/such/file")); Out.ln;
end;

end Defers.

working
done
out
in 2
out
0
-1

defer s; runs s when the program leaves the statements the defer stands in: at their end, or by a return or a break. Deferred statements run the last first, and only those whose defer was reached.

So a file is closed beside the line that opens it, however the procedure ends.

Save a program under its module’s name and run it with raz run Name.raz.
