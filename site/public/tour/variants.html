Enumerations, sets, variants and match

Shapes.raz
module Shapes;
import Out, Fmt, Math;

type
    Colour = (red, green, blue);
    Colours = set of Colour;
    Shape = variant
        circle(r: F64);
        rect(w, h: F64);
        dot;
    end;

proc area(s: Shape): F64;
begin
    match s of
    | circle(r): return Math.pi * r * r;
    | rect(w, h): return w * h;
    | dot: return 0.0;
    end;
end;

proc kind(c: I32): []U8;
begin
    match c of
    | ' ', '\t', '\n': return "space";
    | '0' .. '9': return "digit";
    | 'a' .. 'z', 'A' .. 'Z': return "letter";
    | else: return "other";
    end;
end;

proc main*;
var shapes: [3]Shape; i: INT; warm: Colours; c: Colour;
begin
    shapes := [circle(1.0), rect(2.0, 3.5), dot];
    for i := 0 to 2 do
        Fmt.print("{}\n", [Fmt.real(area(shapes[i]), 2)]);
    end;
    Fmt.print("{} {} {}\n", [Fmt.str(kind('7')), Fmt.str(kind('x')),
        Fmt.str(kind('+'))]);
    warm := {red, blue} - {blue} + {green};
    for c := red to blue do
        if c in warm then Out.int(I64(c)); Out.ln; end;
    end;
end;

end Shapes.

An enumeration is a list of names, in order; a set of it holds any choice of them, in the bits of a word, with in, + for union, * for intersection and - for difference.

A variant holds one of its cases, each with fields of its own, and match is the only way to read them: it runs the arm of the case the variant holds, naming its fields. Every case has an arm, unless an | else: comes last. On an integer, match takes constants and ranges, and must have an | else:.
