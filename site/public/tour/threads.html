Threads

Hits.raz
module Hits;
import Out, Thread;

const Workers = 4;

var
    hits: INT;      (* added to by every thread at once, by XADD *)
    m: Thread.Mutex;
    total: INT;     (* added to under m *)

proc work(arg: ^U8);
var i, n, sum: INT;
begin
    n := INT(arg);
    sum := 0;
    for i := 1 to n do
        sum := sum + i;
        _ := XADD(ADR(hits), 1);
    end;
    Thread.lock(ADR(m));
    total := total + sum;
    Thread.unlock(ADR(m));
end;

proc main*;
var t: [Workers]^Thread.Thread; i: INT; err: ERROR;
begin
    for i := 0 to Workers - 1 do
        t[i], err := Thread.spawn(work, ^U8(1000 * (i + 1)));
        if err # nil then Out.str(MSG(err)); Out.ln; return; end;
    end;
    for i := 0 to Workers - 1 do Thread.join(t[i]); end;
    Out.int(hits); Out.char(' '); Out.int(total); Out.ln;
end;

end Hits.

Thread.spawn runs a procedure in a thread of its own, with a stack of its own, and Thread.join waits for it to end. XADD, CAS and XCHG change a word atomically; a Thread.Mutex keeps a longer section to one thread at a time. Out and Heap lock themselves.

Thread makes its own threads, without C, so a program that uses it can't use C; one that does makes its threads with C's pthread_create instead.
